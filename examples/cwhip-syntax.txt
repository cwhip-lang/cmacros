.define ACTION {int64<int64>}.define RESULT(value) {return(@value)}
# Worker stores a base and adds a value to it.
#
# base stays above public:, so only the class can read it.
# add stays public because start calls it.
class Worker {
    int64 base = 40
    public:
    int64 add(int64 value)
}

# Add value to the worker base.
#
# value is the signed amount to add. Returns base plus value.
int64 Worker.add(int64 value) {
    @RESULT(base + value)
}
